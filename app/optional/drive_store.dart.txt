// Implementazione Drive del backup, tolta dalla build MVP (analisi M7): per
// riattivarla vedi docs/backup.md, sezione "Riattivare Drive".
import 'dart:async';

import 'package:extension_google_sign_in_as_googleapis_auth/extension_google_sign_in_as_googleapis_auth.dart';
import 'package:google_sign_in/google_sign_in.dart';
import 'package:googleapis/drive/v3.dart' as drive;
import 'package:mirabo/data/backup_store.dart';

/// Backup nella cartella nascosta dell'app sul Google Drive personale del
/// genitore (appDataFolder): l'app vede solo i propri file, non il resto del
/// Drive, e il genitore lo trova in Drive › Impostazioni › Gestisci app.
///
/// ID client OAuth passati a build time:
///   --dart-define=GOOGLE_SERVER_CLIENT_ID=<id client Web>   (Android)
///   --dart-define=GOOGLE_IOS_CLIENT_ID=<id client iOS>      (iOS)
class DriveStore implements BackupStore {
  static const _scopes = [drive.DriveApi.driveAppdataScope];
  static const _serverClientId = String.fromEnvironment('GOOGLE_SERVER_CLIENT_ID');
  static const _iosClientId = String.fromEnvironment('GOOGLE_IOS_CLIENT_ID');

  static Future<void>? _initialized;
  GoogleSignInAccount? _account;

  GoogleSignIn get _gsi => GoogleSignIn.instance;

  Future<void> _init() => _initialized ??= _gsi.initialize(
        clientId: _iosClientId.isEmpty ? null : _iosClientId,
        serverClientId: _serverClientId.isEmpty ? null : _serverClientId,
      );

  @override
  Future<String> connect() async {
    await _init();
    final GoogleSignInAccount account;
    try {
      account = await _gsi.authenticate(scopeHint: _scopes);
    } on GoogleSignInException catch (e) {
      if (e.code == GoogleSignInExceptionCode.canceled) throw const BackupCancelled();
      rethrow;
    }
    // Il permesso su Drive si chiede subito, finché il genitore è lì.
    await account.authorizationClient.authorizeScopes(_scopes);
    _account = account;
    return account.email;
  }

  @override
  Future<bool> reconnect() async {
    await _init();
    try {
      _account ??= await _gsi.attemptLightweightAuthentication();
    } on GoogleSignInException {
      return false;
    }
    final account = _account;
    return account != null && await account.authorizationClient.authorizationForScopes(_scopes) != null;
  }

  /// Se l'accesso è scaduto lo richiede: il backup automatico chiama prima
  /// [reconnect], quindi qui arriva solo con il permesso già valido.
  Future<drive.DriveApi> _api() async {
    if (_account == null && !await reconnect()) await connect();
    final client = _account!.authorizationClient;
    final authz = await client.authorizationForScopes(_scopes) ?? await client.authorizeScopes(_scopes);
    return drive.DriveApi(authz.authClient(scopes: _scopes));
  }

  /// Copie presenti nell'appDataFolder: nome -> id, solo quelle di Mirabò.
  Future<Map<String, String>> _files(drive.DriveApi api) async {
    final list = await api.files.list(
      spaces: 'appDataFolder',
      q: "name contains 'mirabo-backup' and trashed = false",
      $fields: 'files(id,name)',
      pageSize: 100,
    );
    return {for (final f in list.files ?? <drive.File>[]) if (f.id != null && f.name != null) f.name!: f.id!};
  }

  /// Carica una copia nuova (non sovrascrive); solo a upload riuscito cancella
  /// le più vecchie oltre [BackupRules.keepCopies].
  @override
  Future<void> upload(List<int> bytes) async {
    final api = await _api();
    final media = drive.Media(Stream.value(bytes), bytes.length, contentType: 'application/json');
    await api.files.create(
      drive.File(name: BackupRules.newName(DateTime.now()), parents: ['appDataFolder']),
      uploadMedia: media,
    );
    try {
      final files = await _files(api);
      for (final name in BackupRules.toPrune(files.keys)) {
        await api.files.delete(files[name]!);
      }
    } catch (_) {
      // La pulizia si ripete al prossimo backup: il nuovo è già al sicuro.
    }
  }

  /// Scarica la copia più recente, con un tetto alla dimensione.
  @override
  Future<List<int>?> download() async {
    final api = await _api();
    final files = await _files(api);
    final newest = BackupRules.newestFirst(files.keys).firstOrNull;
    if (newest == null) return null;
    final media = await api.files.get(files[newest]!, downloadOptions: drive.DownloadOptions.fullMedia) as drive.Media;
    if (media.length != null && media.length! > BackupRules.maxDownloadBytes) {
      throw const FormatException('Backup troppo grande');
    }
    return BackupRules.readLimited(media.stream);
  }

  @override
  Future<void> deleteRemote() async {
    final api = await _api();
    for (final id in (await _files(api)).values) {
      await api.files.delete(id);
    }
  }

  @override
  Future<void> disconnect() async {
    await _init();
    _account = null;
    // disconnect revoca anche il permesso concesso all'app, non solo la sessione.
    await _gsi.disconnect();
  }
}
